const { User } = require("../../backend/db");

async function getUser(id) {

    let user = await User.findOne({ discordId: id });

    if (!user) {
        user = await User.create({ discordId: id });
    }

    return user;
}

async function addCoins(id, amount) {

    const user = await getUser(id);
    user.coins += amount;
    await user.save();
}

module.exports = { getUser, addCoins };
