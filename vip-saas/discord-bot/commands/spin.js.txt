const { getUser, addCoins, addDiamonds } = require("../systems/economy");

module.exports = {
    name: "spin",
    run: async (msg) => {

        const user = getUser(msg.author.id);

        const now = Date.now();
        const cooldown = 24 * 60 * 60 * 1000; // 24h

        if (now - user.lastSpin < cooldown) {
            const timeLeft = cooldown - (now - user.lastSpin);
            const hours = Math.floor(timeLeft / 1000 / 60 / 60);

            return msg.reply(`⏳ Tu jau griezi ratu! Pagaidi ${hours}h`);
        }

        user.lastSpin = now;

        const roll = Math.random();

        if (roll < 0.6) {
            const coins = Math.floor(Math.random() * 500) + 100;
            addCoins(msg.author.id, coins);
            return msg.reply(`🎰 Tu laimēji 🪙 ${coins} coins`);
        }

        if (roll < 0.9) {
            const diamonds = Math.floor(Math.random() * 5) + 1;
            addDiamonds(msg.author.id, diamonds);
            return msg.reply(`💎 Tu laimēji ${diamonds} diamonds`);
        }

        return msg.reply("🔥 JACKPOT! Tu neko nezaudēji, bet gandrīz!");
    }
};
