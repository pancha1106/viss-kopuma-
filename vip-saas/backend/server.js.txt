const express = require("express");
const app = express();

app.use(express.json());

app.get("/status", (req,res) => {
    res.json({ status: "online" });
});

app.post("/vip", (req,res) => {

    console.log("VIP PURCHASE:", req.body.discordId);

    res.sendStatus(200);
});

app.listen(3000, () => {
    console.log("🌐 Backend running on port 3000");
});
